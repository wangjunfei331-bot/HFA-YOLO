class Concat_BiFPN(nn.Module):
    
    def __init__(self, c1, c2):
        super().__init__()  # 最安全写法
        self.align_convs = nn.ModuleList([
            Conv(ch, c2, 1, 1, 0, act=False) if ch != c2 else nn.Identity() 
            for ch in c1
        ])
        # 权重定义
        self.num_inputs = len(c1)
        self.weights = nn.Parameter(torch.ones(self.num_inputs, dtype=torch.float32), requires_grad=True)
        self.epsilon = 1e-4
        # 融合后的处理
        self.conv = Conv(c2, c2, 1, 1, 0) # 融合后建议用 3x3 卷积提取特征
 
    def forward(self, x):
        # x 是一个 tensor 列表
        w = torch.relu(self.weights)
        w = w / (torch.sum(w) + self.epsilon)
        
        # 1. 自动对齐通道 & 2. 加权求和
        out = 0
        for i in range(self.num_inputs):
            # 自动将输入 x[i] 转换为 c2 通道数
            projected_x = self.align_convs[i](x[i])
            out += w[i] * projected_x
            
        return self.conv(out)
